**Goal:** 把 routes.py 里 Planner 的"一次性大 LLM call 编排 N 天"拆成"N 个并发的单天 LLM call"，用 `asyncio.as_completed` 流式 yield `planner.day_done`，目标 wallclock 从 36s 降到 ≤25s（保守 buffer）+ 体感不死寂。

**Architecture:** `Planner` 类新增 `_build_one_day_payload` + `compose_one_day` 两个方法 — 前者抽出单天 prompt，后者 async 调 LLM 返回 `(day_index, DayPlan)`。routes.py 的 compose 段从单次 `_build_compose_payload + llm_call` 改成 `[compose_one_day(d) for d in range(days)]` 用 `as_completed` 流式 yield。每天 LLM 失败降级到 `_synthesize_fallback_route` 单天合成，确保不破坏 v0/v1/v1.5/v2 既有协议。

**Tech Stack:** Python 3.11 + asyncio + pytest（既有）。无新依赖。

        slots_input = [
            {

                "avgprice": p.avgprice,

```python
    async def compose_one_day(
        self,
        day_index: int,
        intent: ParsedIntent,
        template,  # DayTemplate
        anchor: tuple[str, float, float],
        cluster: list[POI],
    ) -> tuple[int, DayPlan]:
        """Run a single LLM call for one day, return (day_index, DayPlan).

        On LLM JSON parse failure or empty stops, falls back to
        _synthesize_fallback_route restricted to this single day.
        """
        payload = self._build_one_day_payload(
            day_index=day_index, intent=intent,
            template=template, anchor=anchor, cluster=cluster,
        )
        try:
            raw = await self.llm_call(self._system_prompt, payload)
            llm_data = json.loads(raw)
        except (json.JSONDecodeError, Exception):
            llm_data = {"stops": []}

        poi_index = {p.openshopid: p for p in cluster}
        stops: list[Stop] = []
        for s in llm_data.get("stops", []) or []:
            pid = s.get("poi_openshopid")
            poi = poi_index.get(pid)
            if poi is None:
                continue
            slot_name = s.get("slot_name", "上午景点")
            slot_def = next(
                (slot for slot in template.slots if slot.name == slot_name),
                template.slots[0],
            )
            stops.append(
                Stop(
                    poi=poi,
                    slot=TimeSlot(
                        name=slot_name, start=slot_def.start, end=slot_def.end
                    ),
                    arrival_time=_parse_time(s.get("arrival_time"), slot_def.start),
                    leave_time=_parse_time(s.get("leave_time"), slot_def.end),
                    transport_to_next_minutes=int(s.get("transport_to_next_minutes", 30)),
                )
            )

        if not stops:
            fallback_days = _synthesize_fallback_route(
                templates=[template], anchors=[anchor], ranked_clusters=[cluster], intent=intent,
            )
            if fallback_days:
                stops = fallback_days[0].stops

        day_plan = DayPlan(
            day_index=day_index,
            anchor_district=anchor[0],
            stops=stops,
        )
        return day_index, day_plan
```

"""SSE protocol tests for v1.6 per-day streaming compose."""

import json


def _parse_events(raw: str) -> list[dict]:
    out = []
    for block in raw.strip().split("\n\n"):
        ev = data = None
        for line in block.split("\n"):
            if line.startswith("event: "):
                ev = line[len("event: "):]
            elif line.startswith("data: "):
                data = json.loads(line[len("data: "):])
        if ev:
            out.append({"event": ev, "data": data})
    return out


def test_per_day_done_count_equals_intent_days(sse_app_client):
    resp = sse_app_client.post(
        "/api/plan/stream",
        json={"free_text": "情侣 3 天深圳预算 3000 爱拍照"},
    )
    assert resp.status_code == 200
    events = _parse_events(resp.text)
    day_dones = [e for e in events if e["event"] == "planner.day_done"]
    assert len(day_dones) == 3


def test_each_day_done_followed_by_rationale_for_same_index(sse_app_client):
    """Each planner.day_done(d) must be immediately followed by
    planner.rationale(stage=compose, day_index=d)."""
    resp = sse_app_client.post(
        "/api/plan/stream",
        json={"free_text": "情侣 3 天深圳预算 3000 爱拍照"},
    )
    events = _parse_events(resp.text)
    seq = [e for e in events if e["event"] in ("planner.day_done", "planner.rationale")]
    seen_day_indices: list[int] = []
    pending_di = None
    for e in seq:
        if e["event"] == "planner.day_done":
            assert pending_di is None, f"prev day_done {pending_di} not paired with rationale"
            pending_di = e["data"]["day_index"]
        elif e["event"] == "planner.rationale" and e["data"].get("stage") == "compose":
            assert pending_di is not None, "rationale arrived before any day_done"
            assert e["data"]["day_index"] == pending_di

def test_single_day_llm_failure_does_not_break_stream(sse_app_client, monkeypatch):
    """If one day's LLM raises, that day falls back to synthesis;
    other days complete normally; stream emits planner.done."""
    from agents import planner as planner_mod

    original = planner_mod.Planner.compose_one_day
    call_counter = {"n": 0}

    async def flaky(self, *, day_index, intent, template, anchor, cluster):
        call_counter["n"] += 1
        if day_index == 1:
            # Force failure on Day 2
            self.llm_call = _raise_llm
        return await original(
            self,
            day_index=day_index, intent=intent,
            template=template, anchor=anchor, cluster=cluster,
        )

    async def _raise_llm(system, user):
        raise RuntimeError("simulated qwen 503")

    monkeypatch.setattr(planner_mod.Planner, "compose_one_day", flaky)

    resp = sse_app_client.post(
        "/api/plan/stream",
        json={"free_text": "情侣 3 天深圳预算 3000 爱拍照"},
    )
    events = _parse_events(resp.text)
    day_dones = [e for e in events if e["event"] == "planner.day_done"]
    assert len(day_dones) == 3  # all 3 days emit even with one LLM fail
    assert any(e["event"] == "planner.done" for e in events)
